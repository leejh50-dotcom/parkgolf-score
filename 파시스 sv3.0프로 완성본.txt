        function fallbackShare(canvas) {
            // 공유 API를 지원하지 않는 경우 이미지 다운로드
            canvas.toBlob(function(blob) {
                const url = URL.createObjectURL(blob);
                const a = document.createElement('a');
                a.href = url;
                a.download = 'parkgolf-result-' + new Date().getTime() + '.png';
                document.body.appendChild(a);
                a.click();
                document.body.removeChild(a);
                URL.revokeObjectURL(url);
                
                document.getElementById('loadingModal').classList.remove('show');
                showToast('이미지가 저장되었습니다!\n갤러리에서 확인 후 공유해주세요.');
            });
        }

        function updateDateTime() {
            const now = new Date();
            const year = now.getFullYear();
            const month = String(now.getMonth() + 1).padStart(2, '0');
            const day = String(now.getDate()).padStart(2, '0');
            const hours = String(now.getHours()).padStart(2, '0');
            const minutes = String(now.getMinutes()).padStart(2, '0');
            const seconds = String(now.getSeconds()).padStart(2, '0');
            const dateStr = year + '.' + month + '.' + day + ' ' + hours + ':' + minutes + ':' + seconds;
            document.getElementById('dateInput').value = '날짜: ' + dateStr;
            document.getElementById('shareDate').textContent = dateStr;
        }
        
        function loadData() {
            const saved = localStorage.getItem('parkGolfData');
            if (saved) {
                const data = JSON.parse(saved);
                
                // 대회명, 팀명 복원
                if (data.tournamentName) {
                    document.getElementById('tournamentName').value = data.tournamentName;
                }
                if (data.teamName) {
                    document.getElementById('teamName').value = data.teamName;
                }
                
                document.getElementById('locationInput').value = data.location || '';
                document.getElementById('shareLocation').textContent = data.location || '미입력';
                selectedHoles = data.selectedHoles || 9;
                coursePars = data.coursePars || coursePars;
                scores = data.scores || scores;
                playerNames = data.playerNames || ['', '', '', ''];
                
                // 이름 복원
                for (let i = 1; i <= 4; i++) {
                    document.getElementById('player' + i).value = playerNames[i-1];
                }
                
                // 소속 복원
                const playerAffiliations = data.playerAffiliations || ['', '', '', ''];
                for (let i = 1; i <= 4; i++) {
                    document.getElementById('affiliation' + i).value = playerAffiliations[i-1] || '';
                }
                
                document.querySelectorAll('.hole-option').forEach(function(el, idx) {
                    el.classList.toggle('active', [9, 18, 27, 36][idx] === selectedHoles);
                });
                
                renderAllCourses();
                showToast('저장된 데이터를 불러왔습니다');
                return true;
            }
            return false;
        }

        function openScoreModal(course, player, hole) {
            currentScoreCourse = course;
            currentScorePlayer = player;
            currentScoreHole = hole;
            const currentValue = scores[course][player][hole];
            if (currentValue) {
                isPenalty = currentValue.includes(')');
                tempScore = currentValue.replace(')', '');
            } else {
                tempScore = '';
                isPenalty = false;
            }
            updateScoreDisplay();
            document.getElementById('scoreModal').classList.add('show');
        }

            if (tempScore.length < 2) {
                tempScore += digit;

            }
        }
        
        function deleteDigit() {

            updateScoreDisplay();
        }

            if (tempScore) {
                isPenalty = !isPenalty;
                updateScoreDisplay();

                display.style.color = '#1e3c72';

        function renderCourseTable(courseName, index) {
            const pars = coursePars[courseName];
            const totalPar = pars.reduce(function(a, b) { return a + b; }, 0);
            const spacer = index > 0 ? '<div style="height: 50px; margin: 0; padding: 0; background: transparent;"></div>' : '';
            
            let html = spacer + '<div class="course-card" style="margin: 0 0 12px 0 !important; padding: 0 !important; width: 100% !important;"><div class="course-header">' + courseName + '코스 (PAR ' + totalPar + ')</div><table class="course-table" style="width: 100% !important; margin: 0 !important;"><thead><tr><th style="width: 30px !important; min-width: 30px !important; max-width: 30px !important; padding: 8px 2px !important;">홀</th><th style="width: 30px !important; min-width: 30px !important; max-width: 30px !important; padding: 8px 2px !important;">파</th><th class="player-header">' + getPlayerName(1) + '</th><th class="player-header">' + getPlayerName(2) + '</th><th class="player-header">' + getPlayerName(3) + '</th><th class="player-header">' + getPlayerName(4) + '</th></tr></thead><tbody>';
            
            for (let hole = 0; hole < 9; hole++) {
                html += '<tr><td class="hole-cell" style="width: 30px !important; min-width: 30px !important; max-width: 30px !important; font-size: 15px !important;">' + (hole + 1) + '</td><td class="par-cell" style="width: 30px !important; min-width: 30px !important; max-width: 30px !important; font-size: 15px !important;" onclick="openParModal(\'' + courseName + '\', ' + hole + ')">' + pars[hole] + '</td>';
                
                for (let player = 1; player <= 4; player++) {
                    const value = scores[courseName][player][hole];
                    const hasPenalty = value.includes(')');
                    const displayValue = value || '-';
                    html += '<td class="score-cell" style="width: calc((100% - 60px) / 4) !important; min-width: 80px !important;"><div class="score-display ' + (hasPenalty ? 'penalty' : '') + '" onclick="openScoreModal(\'' + courseName + '\', ' + player + ', ' + hole + ')">' + displayValue + '</div></td>';
                }
                
                html += '</tr>';
            }
            
            html += '<tr class="total-row"><td style="font-size: 15px !important;">계</td><td style="font-size: 18px !important; font-weight: 900 !important;">' + totalPar + '</td>';
            
            for (let player = 1; player <= 4; player++) {
                const total = scores[courseName][player].reduce(function(sum, score) {
                    const numScore = parseInt(score.replace(')', '')) || 0;
                    return sum + numScore;
                }, 0);
                html += '<td style="font-size: 18px !important; font-weight: 900 !important;">' + (total || '-') + '</td>';
            }
            
            html += '</tr></tbody></table></div>';

        function updateSummary() {
            // 가로형 테이블 업데이트
            for (let player = 1; player <= 4; player++) {
                let totalScore = 0;
                let holeInOneCount = 0;
                let penaltyCount = 0;
                
                // 각 코스별 점수 계산 및 표시
                ['A', 'B', 'C', 'D'].forEach(function(courseName) {
                    const courseTotal = scores[courseName][player].reduce(function(sum, score) {
                        const numScore = parseInt(score.replace(')', '')) || 0;
                        // 홀인원 체크 (점수가 1인 경우)
                        if (numScore === 1) holeInOneCount++;
                        // 벌타 체크 (')' 포함된 경우)
                        if (score.includes(')')) penaltyCount++;
                        return sum + numScore;
                    }, 0);
                    
                    totalScore += courseTotal;
                    
                    // 코스별 점수 표시
                    const scoreCell = document.getElementById('score-' + courseName + '-' + player);
                    if (scoreCell) {
                        scoreCell.textContent = courseTotal || '-';
                    }
                });
                
                // 총계 표시
                const totalCell = document.getElementById('total-new-' + player);
                if (totalCell) {
                    totalCell.textContent = totalScore || '-';
                }
                
                // 홀인원 수 표시
                const holeInOneCell = document.getElementById('holeinone-new-' + player);
                if (holeInOneCell) {
                    holeInOneCell.textContent = holeInOneCount || '-';
                }
                
                // OB/벌타 수 표시
                const penaltyCell = document.getElementById('penalty-new-' + player);
                if (penaltyCell) {
                    penaltyCell.textContent = penaltyCount || '-';
                }
            }
            
            // 순위 계산 및 표시
            let totals = [];
            for (let player = 1; player <= 4; player++) {
                let total = 0;
                ['A', 'B', 'C', 'D'].forEach(function(courseName) {
                    const courseTotal = scores[courseName][player].reduce(function(sum, score) {
                        const numScore = parseInt(score.replace(')', '')) || 0;
                        return sum + numScore;
                    }, 0);
                    total += courseTotal;
                });
                totals.push({player: player, score: total});
            }
            
            // 점수가 있는 플레이어만 순위 계산 (낮은 점수가 더 좋음)
            let validTotals = totals.filter(function(t) { return t.score > 0; });
            validTotals.sort(function(a, b) { return a.score - b.score; });
            
            let ranks = {};
            let currentRank = 1;
            for (let i = 0; i < validTotals.length; i++) {
                if (i > 0 && validTotals[i].score !== validTotals[i-1].score) {
                    currentRank = i + 1;
                }
                ranks[validTotals[i].player] = currentRank;
            }
            
            // 순위 표시
            for (let player = 1; player <= 4; player++) {
                const rankCell = document.getElementById('rank-new-' + player);
                if (rankCell) {
                    if (ranks[player]) {
                        rankCell.textContent = ranks[player] + '위';
                        // 1위는 금색, 2위는 은색, 3위는 동색
                        if (ranks[player] === 1) {
                            rankCell.style.color = '#FFD700';
                        } else if (ranks[player] === 2) {
                            rankCell.style.color = '#C0C0C0';
                        } else if (ranks[player] === 3) {
                            rankCell.style.color = '#CD7F32';
                        } else {
                            rankCell.style.color = '#1e3c72';
                        }
                    } else {
                        rankCell.textContent = '-';
                        rankCell.style.color = '#1e3c72';
                    }
                }
            }
        }

            initDate();
            const hasData = loadData();
            updateHistoryCount();
            
            // 저장된 데이터가 없으면 빈 테이블 렌더링
            if (!hasData) {
                setTimeout(function() {
                    renderAllCourses();
                }, 100);
            }
            
            // 장소 입력 시 결과표에도 실시간 반영
            document.getElementById('locationInput').addEventListener('input', function() {
                document.getElementById('shareLocation').textContent = this.value || '미입력';
            });
            
            for (let i = 1; i <= 4; i++) {
                document.getElementById('player' + i).addEventListener('blur', function() {
                    const name = this.value.trim();
                    if (name) {
                        savePlayerName(name); // 이름 자동 저장
                    }
                    playerNames[i-1] = this.value;
                    renderAllCourses();
                    updateSummary();
                    autoSave();
                });
            }
            
            // 참고 칸 입력 시 자동 저장
            const noteInput = document.getElementById('note-input-new');
            if (noteInput) {
                noteInput.addEventListener('input', function() {
                    // 자동 높이 조절
                    this.style.height = 'auto';
                    this.style.height = this.scrollHeight + 'px';
                    autoSave();
                });
                
                // 초기 높이 설정
                noteInput.style.height = 'auto';
                noteInput.style.height = noteInput.scrollHeight + 'px';
            }
            
            // 모바일에서도 확실하게 작동하도록 추가 이벤트 리스너
            const nameModalCloseBtn = document.querySelector('#nameListModal .modal-close');
            if (nameModalCloseBtn) {
                nameModalCloseBtn.addEventListener('touchend', function(e) {
                    e.preventDefault();
                    closeNameListModal();
                }, { passive: false });
            }

            // 🔥 2. 선수별 상세 점수 데이터 수집
            // ========================================
            
            const playerData = [];
            
            for (let playerNum = 1; playerNum <= 4; playerNum++) {
                const playerName = playerNames[playerNum - 1];

                if (playerName && playerName.trim()) {
                    // 소속 가져오기
                    const affiliationInput = document.getElementById('affiliation' + playerNum);
                    const affiliation = affiliationInput ? affiliationInput.value.trim() : '';
                    const gender = document.getElementById('gender' + playerNum) ? document.getElementById('gender' + playerNum).value : '';
                    
                    // 코스별 점수 계산
                    let scoreA = 0, scoreB = 0, scoreC = 0, scoreD = 0;
                    let totalScore = 0;
                    let holeInOneCount = 0;
                    
                    // A코스
                    if (scores['A'] && scores['A'][playerNum]) {
                        for (let i = 0; i < scores['A'][playerNum].length; i++) {
                            const score = scores['A'][playerNum][i];
                            if (score !== '' && score !== null && score !== undefined) {
                                const numScore = parseInt(score.toString().replace(')', '')) || 0;
                                scoreA += numScore;
                                if (numScore === 1) holeInOneCount++;
                            }
                        }
                    }
                    
                    // B코스
                    if (scores['B'] && scores['B'][playerNum]) {
                        for (let i = 0; i < scores['B'][playerNum].length; i++) {
                            const score = scores['B'][playerNum][i];
                            if (score !== '' && score !== null && score !== undefined) {
                                const numScore = parseInt(score.toString().replace(')', '')) || 0;
                                scoreB += numScore;
                                if (numScore === 1) holeInOneCount++;
                            }
                        }
                    }
                    
                    // C코스
                    if (scores['C'] && scores['C'][playerNum]) {
                        for (let i = 0; i < scores['C'][playerNum].length; i++) {
                            const score = scores['C'][playerNum][i];
                            if (score !== '' && score !== null && score !== undefined) {

                            }

                    // D코스

                        for (let i = 0; i < scores['D'][playerNum].length; i++) {
                            const score = scores['D'][playerNum][i];
                            if (score !== '' && score !== null && score !== undefined) {
                                const numScore = parseInt(score.toString().replace(')', '')) || 0;
                                scoreD += numScore;
                                if (numScore === 1) holeInOneCount++;
                            }
                        }
                    }
                    
                    totalScore = scoreA + scoreB + scoreC + scoreD;
                    
                    // 점수가 0보다 큰 경우만 추가
                    if (totalScore > 0) {
                        playerData.push({
                            name: playerName.trim(),
                            gender: gender,
                            affiliation: affiliation || '',
                            total: totalScore,
                            A: scoreA,

                    },

                });

                } catch (e) {
                    console.log('⚠️ JSON 파싱 실패, 원본 응답:', resultText);
                    // JSON이 아니면 성공으로 간주 (Google Apps Script 리다이렉트 응답)
                    result = { success: true };
                }
                
                // ========================================
                // 🔥 6. 응답 처리 및 메시지
                // ========================================
                
                if (result.success === false) {
                    // 서버에서 에러 반환
                    throw new Error(result.message || '서버 에러 발생');
                }
                
                console.log('✅ 전송 성공!');
                
                const playerNames = playerData.map(p => p.name).join(', ');
                let successMessage = '✅ 자동 전송 완료!\n\n';

        // 텍스트 복사 함수
        function copyScoreText() {
            const teamName = document.getElementById('teamName') ? document.getElementById('teamName').value : '';
            const teamDisplay = teamName || '팀명없음';
            
            const playerData = [];
            
            for (let playerNum = 1; playerNum <= 4; playerNum++) {
                const playerName = playerNames[playerNum - 1];
                
                if (playerName && playerName.trim()) {
                    let totalScore = 0;
                    
                    for (let courseName of courseNames) {
                        if (scores[courseName] && scores[courseName][playerNum]) {
                            const courseScoreArray = scores[courseName][playerNum];
                            for (let i = 0; i < courseScoreArray.length; i++) {
                                const score = courseScoreArray[i];
                                if (score !== '' && score !== null && score !== undefined) {
                                    const numScore = parseInt(score.toString().replace(')', '')) || 0;
                                    totalScore += numScore;
                                }
                            }
                        }
                    }
                    
                    playerData.push({
                        name: playerName.trim(),
                        score: totalScore
                    });
                }
            }
            
            if (playerData.length === 0 || playerData.every(p => p.score === 0)) {
                alert('점수를 먼저 입력해주세요!');
                return;
            }
            
            const lines = [];
            playerData.forEach(p => {
                lines.push(teamDisplay + '\t' + p.name + '\t' + p.score);
            });
            
            const textToCopy = lines.join('\n');
            
            if (navigator.clipboard && navigator.clipboard.writeText) {
                navigator.clipboard.writeText(textToCopy)
                    .then(function() {
                        const previewLines = lines.slice(0, 2);
                        const preview = previewLines.map(line => line.replace(/\t/g, ' | ')).join('\n');
                        const moreText = lines.length > 2 ? '\n... 외 ' + (lines.length - 2) + '명' : '';
                        alert('✅ 복사 완료! (' + lines.length + '명)\n\n' + preview + moreText + '\n\n카톡 단체방에 붙여넣기 하세요!');
                    })
                    .catch(function(error) {
                        showCopyFallback(textToCopy, lines.length);
                    });
            } else {
                showCopyFallback(textToCopy, lines.length);
            }
        }
        
        function showCopyFallback(text, count) {
            const lines = text.split('\n');
            const previewLines = lines.slice(0, 2).map(line => line.replace(/\t/g, ' | ')).join('\n');
            const moreText = lines.length > 2 ? '\n... 외 ' + (lines.length - 2) + '명' : '';
            
            const textarea = document.createElement('textarea');
            textarea.value = text;
            textarea.style.position = 'fixed';
            textarea.style.opacity = '0';
            document.body.appendChild(textarea);
            textarea.select();
            
            try {
                document.execCommand('copy');
                alert('✅ 복사 완료! (' + count + '명)\n\n' + previewLines + moreText + '\n\n카톡 단체방에 붙여넣기 하세요!');
            } catch (err) {
                alert('아래 텍스트를 수동으로 복사해주세요:\n\n' + previewLines + moreText);
            }
            
            document.body.removeChild(textarea);
        }

            // 버튼 클릭 안정화 - iOS 호환 버전
            // ==========================================
            function initButtonListeners() {
                // 디바이스 감지
                const isIOS = /iPad|iPhone|iPod/.test(navigator.userAgent) && !window.MSStream;
                const isMobile = /Android|webOS|iPhone|iPad|iPod|BlackBerry|IEMobile|Opera Mini/i.test(navigator.userAgent);
                
                // 데스크톱용: 강력한 스크롤 방지
                const preventAllDesktop = (e) => {
                    e.preventDefault();
                    e.stopPropagation();
                    e.stopImmediatePropagation();
                    return false;
                };
                
                // 모바일용: 기본 동작만 방지 (클릭은 허용)
                const preventScrollMobile = (e) => {
                    e.stopPropagation();
                    // preventDefault를 호출하지 않아 클릭 이벤트 정상 작동
                };
                
                // 현재 스크롤 위치 저장 및 복원 (데스크톱 전용)
                const executeWithScrollLock = (callback) => {
                    if (!isMobile) {
                        const scrollX = window.scrollX;
                        const scrollY = window.scrollY;
                        
                        requestAnimationFrame(() => {
                            callback();
                            window.scrollTo(scrollX, scrollY);
                        });
                    } else {
                        // 모바일은 그냥 실행
                        callback();
                    }
                };
                
                // 모든 버튼에 공통 스타일 적용
                const allButtons = document.querySelectorAll('button');
                allButtons.forEach(btn => {
                    btn.style.cursor = 'pointer';
                    btn.style.webkitTapHighlightColor = 'rgba(0,0,0,0.1)';
                    
                    // iOS에서 클릭 가능하도록 설정
                    if (isIOS) {
                        btn.style.cursor = 'pointer';
                        // iOS에서 버튼이 클릭 가능하도록 명시
                        btn.addEventListener('touchstart', function() {}, { passive: true });
                    }
                });
                
                // 버튼 이벤트 핸들러 생성 함수
                const createButtonHandler = (callback) => {
                    return function(e) {
                        if (isMobile) {
                            // 모바일: 기본 동작 허용, 함수만 실행
                            e.stopPropagation();
                            setTimeout(() => callback(), 10);
                        } else {
                            // 데스크톱: 스크롤 방지 + 함수 실행
                            preventAllDesktop(e);
                            executeWithScrollLock(() => callback());
                        }
                    };
                };
                
                // 저장하기 버튼
                const saveBtn = document.getElementById('saveBtn');
                if (saveBtn) {
                    saveBtn.addEventListener('click', createButtonHandler(saveData));
                    if (!isMobile) {
                        saveBtn.addEventListener('mousedown', preventAllDesktop);
                    }
                }
                
                // 저장된 기록 버튼
                const historyBtn = document.getElementById('historyBtn');
                if (historyBtn) {
                    historyBtn.addEventListener('click', createButtonHandler(openHistoryModal));
                    if (!isMobile) {
                        historyBtn.addEventListener('mousedown', preventAllDesktop);
                    }
                }
                
                // 점수 지우기 버튼
                const clearBtn = document.getElementById('clearBtn');
                if (clearBtn) {
                    clearBtn.addEventListener('click', createButtonHandler(clearAllScores));
                    if (!isMobile) {
                        clearBtn.addEventListener('mousedown', preventAllDesktop);
                    }
                }
                
                // 텍스트 복사 버튼
                const copyBtn = document.getElementById('copyBtn');
                if (copyBtn) {
                    copyBtn.addEventListener('click', createButtonHandler(copyScoreText));
                    if (!isMobile) {
                        copyBtn.addEventListener('mousedown', preventAllDesktop);
                    }
                }
                
                // 구글시트 자동 전송 버튼
                const autoSendBtn = document.getElementById('autoSendBtn');
                if (autoSendBtn) {
                    // autoSendBtn.addEventListener('click', createButtonHandler(sendToGoogleSheet)); // ⛔ 비활성화
                    if (!isMobile) {
                        // autoSendBtn.addEventListener('mousedown', preventAllDesktop); // ⛔ 비활성화
                    }
                }
                
                // 데스크톱 전용: 스크롤 방지
                if (!isMobile) {
                    let isButtonClicking = false;
                    allButtons.forEach(btn => {
                        btn.addEventListener('mousedown', () => {
                            isButtonClicking = true;
                        });
                        btn.addEventListener('mouseup', () => {
                            setTimeout(() => { isButtonClicking = false; }, 100);
                        });
                    });
                    
                    window.addEventListener('scroll', function(e) {
                        if (isButtonClicking) {
                            e.preventDefault();
                            e.stopPropagation();
                        }
                    }, { passive: false, capture: true });
                }
            }
            
            // 버튼 리스너 초기화
            initButtonListeners();
        });

// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    
    const codeCheck = validateCode(data.code);
    if (!codeCheck.valid) {
      return createResponse(false, codeCheck.message);
    }
    
    const dataCheck = validateData(data);
    if (!dataCheck.valid) {
      return createResponse(false, dataCheck.message);
    }
    
    // ✅ 예선일자에 따라 다른 시트에 저장
    saveToSheet(data);
    
    // ✅ 해당 시트에서 순위 계산
    const sheetName = getSheetName(data.preliminaryDate);
    const sheet = getSheet(sheetName);
    calculateRanks(sheet);
    
    // ✅ 자동 정렬
    autoSortByRank(sheet);
    
    // ✅ 금은동 색깔
    applyRankColors(sheet);
    
    return createResponse(true, '저장 완료 (' + sheetName + ')');
    
  } catch (error) {
    Logger.log('오류: ' + error);
    return createResponse(false, '오류 발생: ' + error.message);
  }
}

function validateCode(code) {
  if (!code) {
    return { valid: false, message: '코드가 없습니다' };
  }
  
  const upperCode = code.toString().trim().toUpperCase();
  const myCode = MY_CODE.toString().trim().toUpperCase();
  
  if (upperCode !== myCode) {
    return { 
      valid: false, 
      message: '코드가 일치하지 않습니다. 관리자에게 확인하세요.' 
    };
  }
  
  return { valid: true };
}

function validateData(data) {
  if (!data.players || data.players.length === 0) {
    return { valid: false, message: '선수 정보가 없습니다' };
  }
  
  if (!data.team) {
    return { valid: false, message: '팀명이 없습니다' };
  }
  
  return { valid: true };
}

// ✅ 예선일자에 따라 시트 이름 결정
function getSheetName(preliminaryDate) {
  if (!preliminaryDate || preliminaryDate === '통합') {
    return '점수기록';
  }
  
  // "1일차" → "점수기록_1일차"
  // "2일차" → "점수기록_2일차"
  // "6일차" → "점수기록_6일차"
  return '점수기록_' + preliminaryDate;
}

function getSheet(sheetName) {
  const ss = SpreadsheetApp.openById(MY_SHEET_ID);
  let sheet = ss.getSheetByName(sheetName);
  
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
    
    const headers = [
      '순위', '코드', '팀명', '선수명', '성별', '소속', '합계',
      'A코스', 'B코스', 'C코스', 'D코스',
      '홀인원', '전송시간', '대회명', '장소', '예선일자'
    ];
    
    sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
    
    const headerRange = sheet.getRange(1, 1, 1, headers.length);
    headerRange.setBackground('#1a73e8');
    headerRange.setFontColor('#ffffff');
    headerRange.setFontWeight('bold');
    headerRange.setHorizontalAlignment('center');
    headerRange.setVerticalAlignment('middle');
    
    for (let i = 1; i <= headers.length; i++) {
      sheet.autoResizeColumn(i);
    }
    
    sheet.setFrozenRows(1);
  }
  
  return sheet;
}

function saveToSheet(data) {
  const sheetName = getSheetName(data.preliminaryDate);
  const sheet = getSheet(sheetName);
  
  const code = data.code || '';
  const team = data.team || '팀명없음';
  const tournament = data.tournament || '대회명없음';
  const location = data.location || '장소없음';
  const preliminaryDate = data.preliminaryDate || '통합';
  const time = data.time || new Date().toLocaleString('ko-KR', {timeZone: 'Asia/Seoul'});
  const players = data.players || [];
  
  for (let i = 0; i < players.length; i++) {
    const p = players[i];
    const row = sheet.getLastRow() + 1;
    
    const rowData = [
      '',
      code,
      team,
      p.name || '',
      p.gender || '',
      p.affiliation || '',
      p.total || 0,
      p.A || 0,
      p.B || 0,
      p.C || 0,
      p.D || 0,
      p.holeInOne || 0,
      time,
      tournament,
      location,
      preliminaryDate
    ];
    
    sheet.getRange(row, 1, 1, rowData.length).setValues([rowData]);
    
    const rowRange = sheet.getRange(row, 1, 1, rowData.length);
    rowRange.setHorizontalAlignment('center');
    rowRange.setVerticalAlignment('middle');
    
    sheet.getRange(row, 6)
      .setBackground('#fff9c4')
      .setFontWeight('bold')
      .setFontSize(11);
  }
}

function calculateRanks(sheet) {
  const lastRow = sheet.getLastRow();
  if (lastRow <= 1) return;
  
  const dataRange = sheet.getRange(2, 1, lastRow - 1, 16);
  const data = dataRange.getValues();
  
  const players = data.map((row, index) => ({
    rowIndex: index + 2,
    code: row[1],
    team: row[2],
    name: row[3],
    gender: row[4],
    affiliation: row[5],
    total: Number(row[6]) || 0,
    A: Number(row[7]) || 0,
    B: Number(row[8]) || 0,
    C: Number(row[9]) || 0,
    D: Number(row[10]) || 0,
    holeInOne: Number(row[11]) || 0
  }));
  
  players.sort((a, b) => {
    if (a.total !== b.total) return a.total - b.total;
    if (a.D !== b.D) return a.D - b.D;
    if (a.C !== b.C) return a.C - b.C;
    if (a.B !== b.B) return a.B - b.B;
    if (a.A !== b.A) return a.A - b.A;
    return 0;
  });
  
  let rank = 1;
  for (let i = 0; i < players.length; i++) {
    if (i > 0) {
      const prev = players[i - 1];
      const curr = players[i];
      
      if (prev.total !== curr.total ||
          prev.D !== curr.D || prev.C !== curr.C ||
          prev.B !== curr.B || prev.A !== curr.A) {
        rank = i + 1;
      }
    }
    players[i].rank = rank;
  }
  
  for (let i = 0; i < players.length; i++) {
    const p = players[i];
    const row = p.rowIndex;
    
    sheet.getRange(row, 1).setValue(p.rank);
  }
}

// ✅ 자동 정렬 함수
function autoSortByRank(sheet) {
  const lastRow = sheet.getLastRow();
  
  if (lastRow <= 1) return;
  
  const range = sheet.getRange(2, 1, lastRow - 1, 15);
  range.sort([
    {column: 1, ascending: true},   // 순위
    {column: 6, ascending: true},   // 합계
    {column: 10, ascending: true}   // D코스
  ]);
}

// ✅ 금은동 색깔 함수
function applyRankColors(sheet) {
  const lastRow = sheet.getLastRow();
  
  if (lastRow <= 1) return;
  
  const allRange = sheet.getRange(2, 1, lastRow - 1, 15);
  allRange.setBackground('#ffffff');
  allRange.setFontWeight('normal');
  allRange.setFontSize(10);
  allRange.setFontColor('#000000');
  
  const rankData = sheet.getRange(2, 1, lastRow - 1, 15).getValues();
  
  for (let i = 0; i < rankData.length; i++) {
    const rank = rankData[i][0];
    const rowNum = i + 2;
    const rowRange = sheet.getRange(rowNum, 1, 1, 15);
    
    if (rank === 1) {
      rowRange.setBackground('#ffd700');
      rowRange.setFontWeight('bold');
      rowRange.setFontSize(11);
      rowRange.setFontColor('#000000');
    } else if (rank === 2) {
      rowRange.setBackground('#c0c0c0');
      rowRange.setFontWeight('bold');
      rowRange.setFontSize(11);
      rowRange.setFontColor('#000000');
    } else if (rank === 3) {
      rowRange.setBackground('#cd7f32');
      rowRange.setFontWeight('bold');
      rowRange.setFontSize(11);
      rowRange.setFontColor('#ffffff');
    }
    
    sheet.getRange(rowNum, 6)
      .setBackground('#fff9c4')
      .setFontWeight('bold')
      .setFontSize(11)
      .setFontColor('#000000');
    
    const holeInOne = rankData[i][10];
    if (holeInOne > 0) {
      sheet.getRange(rowNum, 11)
        .setBackground('#ffcccc')
        .setFontWeight('bold')
        .setFontColor('#cc0000');
    }
  }
}

function createResponse(success, message) {
  return ContentService
    .createTextOutput(JSON.stringify({
      success: success,
      message: message
    }))
    .setMimeType(ContentService.MimeType.JSON);
}

function clearAllSheets() {
  const sheetNames = [
    '점수기록',
    '점수기록_1일차',
    '점수기록_2일차',
    '점수기록_3일차',
    '점수기록_4일차',
    '점수기록_5일차',
    '점수기록_6일차'
  ];
  
  const ss = SpreadsheetApp.openById(MY_SHEET_ID);
  
  sheetNames.forEach(name => {
    const sheet = ss.getSheetByName(name);
    if (sheet) {
      const lastRow = sheet.getLastRow();
      if (lastRow > 1) {
        sheet.deleteRows(2, lastRow - 1);
        Logger.log('✅ ' + name + ' 초기화 완료');
      }
    }
  });
  
  Logger.log('');
  Logger.log('🎉 모든 시트 초기화 완료!');
}

function copyMasterTemplate() {
    const urlInput = document.getElementById('masterTemplateUrl');
    const url = urlInput.value.trim();
    
    // URL 검증
    if (!url) {
        alert('⚠️ 마스터 템플릿 URL을 입력해주세요!');
        urlInput.focus();
        return;
    }
    
    // Google Sheets URL 검증
    if (!url.includes('docs.google.com/spreadsheets')) {
        alert('❌ 올바른 구글 시트 URL이 아닙니다!\n\n' +
              '예시: https://docs.google.com/spreadsheets/d/...');
        urlInput.focus();
        return;
    }
    
    // 시트 ID 추출
    let sheetId = '';
    
    // 패턴 1: /d/[ID]/edit
    const pattern1 = /\/d\/([a-zA-Z0-9-_]+)/;
    const match1 = url.match(pattern1);
    
    if (match1) {
        sheetId = match1[1];
    } else {
        alert('❌ URL에서 시트 ID를 찾을 수 없습니다.\n\n' +
              '올바른 URL 형식:\n' +
              'https://docs.google.com/spreadsheets/d/[시트ID]/edit');
        return;
    }
    
    // 복사 URL 생성
    const copyUrl = `https://docs.google.com/spreadsheets/d/${sheetId}/copy`;
    
    // 확인 메시지
    const confirmed = confirm(
        '📋 마스터 템플릿 복사\n\n' +
        '✅ 새 창에서 복사본이 생성됩니다.\n\n' +
        '복사 후 할 일:\n' +
        '1. 복사본 이름 변경 (예: 2026-01-28 신년대회)\n' +
        '2. Apps Script에서 코드 설정\n' +
        '3. 배포 및 QR 생성\n\n' +
        '계속하시겠습니까?'
    );
    
    if (!confirmed) {
        return;
    }
    
    // localStorage에 URL 저장 (다음에 자동 입력)
    localStorage.setItem('parkgolf_master_template_url', url);
    
    // 새 창에서 복사 URL 열기
    const newWindow = window.open(copyUrl, '_blank');
    
    if (newWindow) {
        alert('✅ 복사 완료!\n\n' +
              '새 창에서 복사본이 열렸습니다.\n\n' +
              '다음 단계:\n' +
              '1. 파일 이름을 대회명으로 변경\n' +
              '2. 확장 프로그램 → Apps Script\n' +
              '3. 아래 코드를 복사해서 붙여넣기\n' +
              '4. MY_CODE 수정 후 배포');
        
        // Apps Script 섹션으로 스크롤
        const codeSection = document.querySelector('#appsScriptSection');
        if (codeSection) {
            setTimeout(() => {
                codeSection.scrollIntoView({ behavior: 'smooth', block: 'center' });
            }, 500);
        }
    } else {
        alert('⚠️ 팝업이 차단되었습니다!\n\n' +
              '브라우저 설정에서 팝업을 허용하거나\n' +
              '아래 URL을 직접 열어주세요:\n\n' +
              copyUrl);
    }
}

function autoAssignTeams() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // 접수명단 시트 가져오기
  const sourceSheet = ss.getSheetByName('접수명단');
  if (!sourceSheet) {
    SpreadsheetApp.getUi().alert('❌ "접수명단" 시트를 찾을 수 없습니다!\\n\\n시트 이름을 확인해주세요.');
    return;
  }
  
  // 데이터 읽기
  const data = sourceSheet.getDataRange().getValues();
  if (data.length <= 1) {
    SpreadsheetApp.getUi().alert('❌ 접수 명단이 없습니다!');
    return;
  }
  
  // 헤더 제거
  const players = data.slice(1);
  
  // 랜덤 섞기 (Fisher-Yates shuffle)
  for (let i = validPlayers.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [validPlayers[i], validPlayers[j]] = [validPlayers[j], validPlayers[i]];
  }
  
  // 조편성 결과 시트 생성 또는 초기화
  let resultSheet = ss.getSheetByName('조편성결과');
  if (resultSheet) {
    ss.deleteSheet(resultSheet);
  }
  resultSheet = ss.insertSheet('조편성결과');
  
  // 헤더 작성
  resultSheet.appendRow(['조', '이름', '성별', '생년', '소속', '연락처']);
  const headerRange = resultSheet.getRange(1, 1, 1, 6);
  headerRange.setBackground('#f093fb')
             .setFontColor('#FFFFFF')
             .setFontWeight('bold')
             .setHorizontalAlignment('center');
  
  // 4명 1조로 배치
  let teamNum = 1;
  for (let i = 0; i < validPlayers.length; i += 4) {
    const teamSize = Math.min(4, validPlayers.length - i);
    
    for (let j = 0; j < teamSize; j++) {
      const player = validPlayers[i + j];
      resultSheet.appendRow([
        teamNum + '조',
        player[1] || '',  // 이름
        player[2] || '',  // 성별
        player[3] || '',  // 생년
        player[4] || '',  // 소속
        player[5] || ''   // 연락처
      ]);
    }
    
    // 조 구분선
    if (i + 4 < validPlayers.length) {
      const separatorRow = resultSheet.getLastRow();
      const separatorRange = resultSheet.getRange(separatorRow, 1, 1, 6);
      separatorRange.setBorder(false, false, true, false, false, false, '#e0e0e0', SpreadsheetApp.BorderStyle.SOLID_MEDIUM);
    }
    
    teamNum++;
  }
  
  // 자동 정렬
  const dataRange = resultSheet.getRange(2, 1, resultSheet.getLastRow() - 1, 6);
  dataRange.sort([{column: 1, ascending: true}]);
  
  // 열 너비 자동 조정
  resultSheet.autoResizeColumns(1, 6);
  
  // 성공 메시지
  const totalTeams = Math.ceil(validPlayers.length / 4);
  SpreadsheetApp.getUi().alert(
    '✅ 조편성 완료!\\n\\n' +
    '총 인원: ' + validPlayers.length + '명\\n' +
    '총 조 수: ' + totalTeams + '개 조\\n\\n' +
    '"조편성결과" 시트를 확인하세요!'
  );
}`;

async function copyAppsScript() {
    console.log('📋 복사 시작, 코드 길이:', APPS_SCRIPT_CODE.length);
    
    // 방법 1: Clipboard API (최신, 가장 확실)

            console.log('✅ Clipboard API 복사 성공');
            return;
        } catch (err) {
            console.log('⚠️ Clipboard API 실패:', err);
            // 다음 방법 시도
        }
    }
    
    // 방법 2: execCommand (구형 브라우저 지원)
    try {
        const textArea = document.createElement('textarea');
        textArea.value = APPS_SCRIPT_CODE;
        textArea.style.position = 'fixed';
        textArea.style.top = '0';
        textArea.style.left = '0';
        textArea.style.opacity = '0';
        textArea.style.pointerEvents = 'none';
        document.body.appendChild(textArea);
        
        textArea.focus();
        textArea.select();
        
        const successful = document.execCommand('copy');
        document.body.removeChild(textArea);
        
        if (successful) {
            alert('✅ Apps Script 코드가 복사되었습니다!\n\n코드 길이: ' + APPS_SCRIPT_CODE.length + ' 글자\n\n구글 Apps Script 창에 붙여넣기(Ctrl+V)하세요.');
            console.log('✅ execCommand 복사 성공');
            return;
        } else {
            throw new Error('execCommand 실패');
        }
    } catch (err) {
        console.error('⚠️ execCommand 실패:', err);
        // 마지막 방법 시도
    }
    
    // 방법 3: 모달 창 (수동 복사)
    console.log('ℹ️ 수동 복사 모달 표시');
    showCodeModal();
}
