// ---------- 유틸 ----------
function num2(v) {
  const n = parseInt((v || '').toString().replace(/[^0-9]/g,''), 10);
  return isNaN(n) ? 0 : n;
}
function padDash(v) {
  return (v===0) ? '0' : (v ? v : '-');
}

function initDate() {
  updateDateTime();

function updateDateTime() {
  const now = new Date();
  const year = now.getFullYear();
  const month = String(now.getMonth() + 1).padStart(2, '0');
  const day = String(now.getDate()).padStart(2, '0');
  const hours = String(now.getHours()).padStart(2, '0');
  const minutes = String(now.getMinutes()).padStart(2, '0');
  const seconds = String(now.getSeconds()).padStart(2, '0');
  const dateStr = year + '.' + month + '.' + day + ' ' + hours + ':' + minutes + ':' + seconds;
  const dateInput = document.getElementById('dateInput');
  if (dateInput) dateInput.value = '날짜: ' + dateStr;
  const shareDate = document.getElementById('shareDate');
  if (shareDate) shareDate.textContent = dateStr;
}

// ---------- 점수 입력 바인딩 ----------
function bindScoreInputs() {
  ['A','B','C','D'].forEach(function(c) {
    for (let i=1;i<=4;i++) {
      const el = document.getElementById('in-'+c+'-'+i);
      if (!el) continue;
      el.addEventListener('input', function() {
        // 숫자만, 최대 2자리
        const cleaned = this.value.replace(/[^0-9]/g,'').slice(0,2);
        this.value = cleaned;
        summaryScores[c][i] = cleaned;
        updateTotalsAndRanks();
        autoSave();
      });
      el.addEventListener('blur', function() {
        // 빈칸은 유지(표시는 placeholder)
        autoSave();
      });
    }
  });
}

function updateTotalsAndRanks() {
  // 총계 계산
  const totals = [];
  for (let i=1;i<=4;i++) {
    const t = num2(summaryScores.A[i]) + num2(summaryScores.B[i]) + num2(summaryScores.C[i]) + num2(summaryScores.D[i]);
    totals[i-1] = t;
    const td = document.getElementById('total-new-'+i);
    if (td) td.textContent = (t>0) ? String(t) : '-';
  }

  // 순위(점수 낮을수록 1위). 0점(미입력)은 순위 '-'
  const players = totals.map((t, idx) => ({idx: idx+1, total: t})).filter(p => p.total>0);
  players.sort((a,b)=> a.total-b.total);
  // 동점 처리: 같은 점수면 같은 순위
  const rankMap = {};
  let rank=1;
  for (let j=0;j<players.length;j++) {
    if (j>0 && players[j].total !== players[j-1].total) rank = j+1;
    rankMap[players[j].idx] = rank;
  }
  for (let i=1;i<=4;i++) {
    const td = document.getElementById('rank-new-'+i);
    if (!td) continue;
    td.textContent = rankMap[i] ? (rankMap[i] + '위') : '-';
  }
}

  }
    if (data.tournamentName) document.getElementById('tournamentName').value = data.tournamentName;

    if (ss) {

// ---------- 이름 저장/불러오기(기존 기능 유지) ----------
function savePlayerName(name, gender, affiliation) {
  if (!name || name.trim() === '') return;
  const key = 'parkGolfPlayerNames';
  let saved = JSON.parse(localStorage.getItem(key) || '[]');

  // 구버전(문자열 배열) 호환
  if (saved.length && typeof saved[0] === 'string') {
    saved = saved.map(n => ({ name: n, gender: '', affiliation: '' }));
  }

  const idx = saved.findIndex(p => (p && p.name) === name);
  const newObj = { name: name, gender: (gender||'').toString(), affiliation: (affiliation||'').toString() };
  if (idx >= 0) saved[idx] = newObj;
  else saved.unshift(newObj);

  // 최대 200개 제한
  if (saved.length > 200) saved = saved.slice(0,200);
  localStorage.setItem(key, JSON.stringify(saved));
}

function applySavedName(name, gender, affiliation) {
  // 빈 슬롯(1~4) 중 첫번째에 적용
  for (let i=1;i<=4;i++) {
    const inp = document.getElementById('player'+i);
    if (inp && !(inp.value || '').trim()) {
      inp.value = name || '';
      const gSel = document.getElementById('gender'+i);
      if (gSel) gSel.value = gender || '';
      document.getElementById('affiliation'+i).value = affiliation || '';
      break;
    }
  }
  syncNameHeaders();
  autoSave();
}
function clearPlayerNames() {
  if (!confirm('이름/성별/소속을 모두 지우시겠습니까?')) return;
  for (let i=1;i<=4;i++) {
    document.getElementById('player'+i).value = '';
    const gSel = document.getElementById('gender'+i);
    if (gSel) gSel.value = '';
    document.getElementById('affiliation'+i).value = '';
  }
  syncNameHeaders();
  autoSave();
  showToast('이름이 지워졌습니다');
}
function escapeHtml(s) {
  return (s || '').toString()
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;');
}

// ---------- 점수 초기화 ----------
function clearScores() {
  if (!confirm('코스별 점수를 모두 지우시겠습니까?')) return;
  ['A','B','C','D'].forEach(function(c) {
    for (let i=1;i<=4;i++) {
      summaryScores[c][i] = '';
      const el = document.getElementById('in-'+c+'-'+i);
      if (el) el.value = '';
    }
  });
  updateTotalsAndRanks();
  autoSave();
  showToast('점수가 지워졌습니다');
}

// ---------- 텍스트 복사 ----------
function copyAsText() {
  const location = document.getElementById('locationInput').value || '';
  const dateStr = document.getElementById('shareDate').textContent || '';
  const lines = [];
  lines.push('파크골프 경기 결과');
  lines.push('날짜: ' + dateStr);
  lines.push('장소: ' + (location || '미입력'));
  lines.push('');
  lines.push('코스별 합계');
  lines.push('이름: ' + [1,2,3,4].map(i => (document.getElementById('player'+i).value || ('경기자'+i))).join(', '));
  ['A','B','C','D'].forEach(function(c) {
    const row = [c].concat([1,2,3,4].map(i => (document.getElementById('in-'+c+'-'+i).value || '-')));
    lines.push(row.join('  '));
  });
  lines.push('총계  ' + [1,2,3,4].map(i => (document.getElementById('total-new-'+i).textContent || '-')).join('  '));
  lines.push('순위  ' + [1,2,3,4].map(i => (document.getElementById('rank-new-'+i).textContent || '-')).join('  '));

  const text = lines.join('\n');
  navigator.clipboard.writeText(text).then(function() {
    showToast('텍스트가 복사되었습니다');
  }).catch(function() {
    // 폴백
    const ta = document.createElement('textarea');
    ta.value = text;
    document.body.appendChild(ta);
    ta.select();
    document.execCommand('copy');
    document.body.removeChild(ta);
    showToast('텍스트가 복사되었습니다');
  });
}

async function sendToGoogleSheet() {
            // ========================================
            // 🔥 1. 필수 정보 수집
            // ========================================
            
            // 대회 코드 (localStorage에서 가져오기)
            const tournamentCode = localStorage.getItem('parkgolf_tournament_code') || '';
            if (!tournamentCode) {
                alert('⚠️ 대회 코드가 없습니다!\n\nQR 코드를 다시 스캔하거나 대회 코드를 입력해주세요.');
                return;
            }
            
            // 웹앱 URL (localStorage에서 가져오기)
            const webAppUrl = localStorage.getItem('parkgolf_webapp_url') || '';
            if (!webAppUrl) {
                alert('⚠️ 웹 앱 URL이 없습니다!\n\nQR 코드를 다시 스캔해주세요.');
                return;
            }
            
            // 예선일자 (localStorage에서 가져오기, 없으면 '통합')
            const preliminaryDate = localStorage.getItem('parkgolf_preliminary_date') || '통합';
            
            // 대회명과 팀명
            const tournamentName = document.getElementById('tournamentName') ? document.getElementById('tournamentName').value.trim() : '';
            const tournamentDisplay = tournamentName || '대회명없음';
            
            const teamName = document.getElementById('teamName') ? document.getElementById('teamName').value.trim() : '';
            const teamDisplay = teamName || '팀명없음';
            
            // 장소
            const location = document.getElementById('locationInput') ? document.getElementById('locationInput').value.trim() : '';
            const locationDisplay = location || '장소없음';
            
            // ========================================
            // 🔥 2. 선수별 상세 점수 데이터 수집
            // ========================================
            
            const playerData = (window.playerData && Array.isArray(window.playerData)) ? window.playerData : (typeof buildPlayerDataForSend_ === 'function' ? buildPlayerDataForSend_() : []);
            
            // ✅ SV4.0 간소화: 코스별 합계표(직접입력) 기반 전송
// ========================================
            // 🔥 3. 데이터 검증
            // ========================================
            
            if (playerData.length === 0) {
                alert('⚠️ 점수를 먼저 입력해주세요!');
                return;
            }

            // ✅ 성별 미선택 전송 차단 (누락 방지)
            for (let i = 0; i < playerData.length; i++) {
                if (!playerData[i].gender) {
                    const pNum = playerData[i].playerNum || (i + 1);
                    alert('⚠️ 성별이 선택되지 않았습니다!\n\n' + playerData[i].name + ' 선수의 성별(남/여)을 선택한 후 전송하세요.');
                    const sel = document.getElementById('gender' + pNum);
                    if (sel) {
                        sel.focus();
                        try { sel.scrollIntoView({ behavior: 'smooth', block: 'center' }); } catch (e) {}
                    }
                    return;
                }
            }

            // ✅ 재전송 완전 차단(클라이언트 1차 방어): 같은 조 + 같은 이름 조합
            const normalizedNames = playerData
                .map(p => (p.name || '').toString().trim())
                .filter(Boolean)
                .sort()
                .join(',');
            const groupKey = ['SV3_SENT', tournamentCode.toString().trim().toUpperCase(), preliminaryDate.toString().trim(), teamDisplay.toString().trim(), normalizedNames].join('|');
            let sentGroups = [];
            try {
                sentGroups = JSON.parse(localStorage.getItem('sv3_sent_groups') || '[]');
                if (!Array.isArray(sentGroups)) sentGroups = [];
            } catch (e) {
                sentGroups = [];
            }
            if (sentGroups.includes(groupKey)) {
                alert('⛔ 이미 전송된 조/선수 조합입니다.\n\n전송 성공 후에는 같은 조/같은 이름 조합으로 재전송(수정)할 수 없습니다.\n\n수정이 필요하면 관리자에게 문의 하세요.');
                return;
            }

            // ✅ 전송 전 경고/확인 메시지
            let confirmMsg = '📤 전송 확인(주의!)\n\n';
            confirmMsg += '대회코드: ' + tournamentCode + '\n';
            confirmMsg += '대회명: ' + tournamentDisplay + '\n';
            confirmMsg += '팀명(조명): ' + teamDisplay + '\n';
            if (preliminaryDate && preliminaryDate !== '통합') confirmMsg += '예선일자: ' + preliminaryDate + '\n';
            confirmMsg += '인원: ' + playerData.length + '명\n';
            confirmMsg += '선수: ' + playerData.map(p => p.name).join(', ') + '\n\n';
            confirmMsg += '⚠️ 전송 후에는 수정/재전송이 불가능합니다.\n';
            confirmMsg += '구글 시트로 전송하시겠습니까?';
            if (!confirm(confirmMsg)) {
                return;
            }
            
            // ========================================
            // 🔥 4. 전송 준비
            // ========================================
            
            const btn = document.getElementById('autoSendBtn');
            const originalText = btn.innerHTML;
            btn.innerHTML = '⏳ 전송 중...';
            btn.disabled = true;
            
            // 전송할 데이터 구조
            const payload = {
                code: tournamentCode,
                tournament: tournamentDisplay,
                team: teamDisplay,
                location: locationDisplay,
                preliminaryDate: preliminaryDate,
                time: new Date().toLocaleString('ko-KR', {timeZone: 'Asia/Seoul'}),
                players: playerData
            };
            
            console.log('📤 전송 데이터:', payload);
            
            try {
                // ========================================
                // 🔥 5. 구글 시트로 전송
                // ========================================
                
                console.log('📤 전송 시작...');
                
                const response = await fetch(webAppUrl, {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'text/plain'  // Google Apps Script 권장
                    },
                    body: JSON.stringify(payload)
                });
                
                console.log('📥 응답 수신:', response.status);
                
                // 응답 읽기
                const resultText = await response.text();
                const resultJson = safeParseJson_(resultText);
                if (!response.ok) {
                    const msg = (resultJson && resultJson.message) ? resultJson.message : (resultText || ('HTTP ' + response.status));
                    throw new Error(msg);
                }
                if (resultJson && resultJson.success === false) {
                    throw new Error(resultJson.message || '서버에서 저장을 거부했습니다.');
                }

                console.log('📄 응답 내용:', resultText);
                
                let result;
                try {
                    result = JSON.parse(resultText);
                } catch (e) {
                    console.log('⚠️ JSON 파싱 실패, 원본 응답:', resultText);
                    // JSON이 아니면 성공으로 간주 (Google Apps Script 리다이렉트 응답)
                    result = { success: true };
                }
                
                // ========================================
                // 🔥 6. 응답 처리 및 메시지
                // ========================================
                
                if (result.success === false) {
                    // 서버에서 에러 반환
                    throw new Error(result.message || '서버 에러 발생');
                }
                
                console.log('✅ 전송 성공!');

                // ✅ 전송 성공 → 재전송 완전 차단 키 저장(로컬)
                try {
                    if (!sentGroups.includes(groupKey)) {
                        sentGroups.push(groupKey);
                        localStorage.setItem('sv3_sent_groups', JSON.stringify(sentGroups));
                    }
                } catch (e) {}
                
                const playerNames = playerData.map(p => p.name).join(', ');
                let successMessage = '✅ 자동 전송 완료!\n\n';
                successMessage += '대회코드: ' + tournamentCode + '\n';
                successMessage += '대회명: ' + tournamentDisplay + '\n';
                successMessage += '팀명: ' + teamDisplay + '\n';
                if (preliminaryDate && preliminaryDate !== '통합') {
                    successMessage += '예선일자: ' + preliminaryDate + '\n';
                }
                successMessage += '인원: ' + playerData.length + '명\n';
                successMessage += '선수: ' + playerNames + '\n\n';
                
                // 중복 차단 메시지가 있으면 표시
                if (result.message && result.message.includes('중복')) {
                    successMessage += '⚠️ ' + result.message;
                } else {
                    successMessage += '구글 시트를 확인해주세요!';
                }
                
                alert(successMessage);
                
            } catch (error) {
                console.error('❌ 전송 오류:', error);
                const msg = (error && error.message) ? error.message : String(error);

                // 서버가 "이미 전송"을 알려주면 로컬에도 저장하여 즉시 차단
                if (msg && msg.includes('이미 전송')) {
                    try {
                        if (!sentGroups.includes(groupKey)) {
                            sentGroups.push(groupKey);
                            localStorage.setItem('sv3_sent_groups', JSON.stringify(sentGroups));
                        }
                    } catch (e) {}
                    alert('⛔ ' + msg);
                } else {
                    alert('⚠️ 자동 전송 실패\n\n' +
                          (msg ? (msg + '\n\n') : '') +
                          '인터넷 연결을 확인하거나\n"텍스트 복사" 버튼을 이용해주세요.');
                }
            } finally {
                // ========================================
                // 🔥 7. 버튼 복원
                // ========================================
                btn.innerHTML = originalText;
                btn.disabled = false;
            }
        }

// ---------- SV4 전송용 데이터 생성(기존 sendToGoogleSheet 내부 playerData 계산 구간 대체) ----------
function buildPlayerDataForSend_() {
  const playerData = [];
  for (let playerNum = 1; playerNum <= 4; playerNum++) {
    const playerName = (document.getElementById('player' + playerNum).value || '').trim();
    if (!playerName) continue;

    const affInput = document.getElementById('affiliation' + playerNum);
    const affiliation = affInput ? (affInput.value || '').trim() : '';
    const gender = document.getElementById('gender' + playerNum) ? (document.getElementById('gender' + playerNum).value || '') : '';

    const scoreA = num2(summaryScores.A[playerNum]);
    const scoreB = num2(summaryScores.B[playerNum]);
    const scoreC = num2(summaryScores.C[playerNum]);
    const scoreD = num2(summaryScores.D[playerNum]);
    const totalScore = scoreA + scoreB + scoreC + scoreD;

    if (totalScore > 0) {
      playerData.push({
        playerNum: playerNum,
        name: playerName,
        gender: gender,
        affiliation: affiliation || '',
        total: totalScore,
        A: scoreA,
        B: scoreB,
        C: scoreC,
        D: scoreD,
        holeInOne: 0
      });
    }
  }
  return playerData;
}

// ---------- QR(쿼리스트링) 자동 적용 ----------
// QR 생성기에서 만든 링크: ?code=...&url=...&preliminaryDate=...
function initFromQueryParams() {
  try {
    const params = new URLSearchParams(window.location.search || '');
    const code = params.get('code');
    const url  = params.get('url');
    const pre  = params.get('preliminaryDate');

    let changed = false;

    if (code) {
      localStorage.setItem('parkgolf_tournament_code', code);
      const inp = document.getElementById('tournamentCode');
      if (inp) inp.value = code;
      
    // QR로 새 코드가 들어오면, 기존 저장 데이터(parkGolfData_sv4)에도 즉시 반영(구코드 덮어쓰기 방지)
    try {
      const saved = localStorage.getItem('parkGolfData_sv4');
      if (saved) {
        const obj = JSON.parse(saved);
        obj.tournamentCode = code;
        localStorage.setItem('parkGolfData_sv4', JSON.stringify(obj));
      }
    } catch (e) {}

      changed = true;
    }

    if (url) {
      const decodedUrl = decodeURIComponent(url);
      localStorage.setItem('parkgolf_webapp_url', decodedUrl);
      changed = true;
    }

    if (pre) {
      localStorage.setItem('parkgolf_preliminary_date', pre);
      changed = true;
    }

    // QR에서 넘어온 값이 없으면(직접 접속) 기존 저장값을 입력칸에 반영
    if (!code) {
      const savedCode = localStorage.getItem('parkgolf_tournament_code') || '';
      const inp = document.getElementById('tournamentCode');
      if (inp && savedCode && !inp.value) inp.value = savedCode;
    }

    if (changed) {
      showToast('✅ QR 정보(코드/URL)가 적용되었습니다');
    }
  } catch (e) {
    console.log('initFromQueryParams error:', e);
  }
}

    const myCode = MY_CODE.toString().trim().toUpperCase();
    if (incomingCode !== myCode) {
      throw new Error('대회 코드가 일치하지 않습니다.\n\n입력된 코드: [' + incomingCode + ']\n설정된 코드: [' + myCode + ']');
    }

    const sheetName = getSheetName(preliminaryDate);
    console.log('📋 대상 시트:', sheetName);

    const result = insertOnlyToSheet_(sheetName, incomingCode, tournament, team, location, preliminaryDate, timeStr, players);

    return ContentService
      .createTextOutput(JSON.stringify({
        success: true,
        message: '전송 완료! (신규 ' + result.inserted + '건)\n※ 어떤 경우라도 재전송(수정) 불가'
      }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    console.error('❌ 에러:', err && err.toString ? err.toString() : err);
    return ContentService
      .createTextOutput(JSON.stringify({
        success: false,
        message: (err && err.message) ? err.message : String(err)
      }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function getSheetName(preliminaryDate) {
  if (!preliminaryDate || preliminaryDate === '통합') return '점수기록';
  if (preliminaryDate === '본선') return '점수기록_본선';
  return '점수기록_' + preliminaryDate;
}

/**
 * ✅ 재전송(수정) 절대 금지 모드
 * - 기존 행이 있으면 즉시 에러(저장 거부)
 * - 없으면 appendRow로만 추가
 */
function insertOnlyToSheet_(sheetName, code, tournament, team, location, preliminaryDate, timeStr, players) {
  const ss = getSpreadsheet_();
  let sheet = ss.getSheetByName(sheetName);
  if (!sheet) sheet = createNewSheet_(ss, sheetName);

  // ✅ 기존 데이터 인덱스 맵(중복 체크)
  const lastRow = sheet.getLastRow();
  const existingMap = new Map();

  if (lastRow >= 2) {
    const values = sheet.getRange(2, 1, lastRow - 1, 16).getValues(); // A~P
    for (let i = 0; i < values.length; i++) {
      const row = values[i];
      const exCode = (row[1] || '').toString().trim();   // B
      const exTeam = (row[2] || '').toString().trim();   // C
      const exName = (row[3] || '').toString().trim();   // D
      const exPre  = (row[15] || '').toString().trim();  // P
      if (!exCode || !exName) continue;
      const key = [exCode, exTeam, exName, exPre].join('|');
      existingMap.set(key, i + 2);
    }
  }

  let inserted = 0;

  for (let i = 0; i < players.length; i++) {
    const p = players[i] || {};
    const name = (p.name || '').toString().trim();
    if (!name) continue;

    const gender = (p.gender || p.sex || '').toString().trim();
    const affiliation = (p.affiliation || p.affil || p.org || '').toString().trim();

    // ✅ 서버 안전장치: 성별 누락이면 저장 거부
    if (!gender) {
      throw new Error('성별이 비어 있습니다. SV3.0에서 남/여 선택 후 다시 시도하세요.');
    }

    const key = [code, team, name, preliminaryDate].join('|');
    if (existingMap.has(key)) {
      // 🚫 재전송(수정) 금지
      throw new Error('이미 전송된 기록입니다.\n수정이 필요하면 관리자에게 문의 하세요.');
    }

    const rowValues = [
      '',               // A 순위
      code,             // B 대회코드
      team,             // C 조명
      name,             // D 선수명
      gender,           // E 성별
      affiliation,      // F 소속
      num_(p.total),    // G 합계
      num_(p.A),        // H A코스
      num_(p.B),        // I B코스
      num_(p.C),        // J C코스
      num_(p.D),        // K D코스
      num_(p.holeInOne),// L 홀인원
      timeStr,          // M 전송시간
      tournament,       // N 대회명
      location,         // O 장소
      preliminaryDate   // P 예선일자
    ];

    sheet.appendRow(rowValues);
    inserted++;
    existingMap.set(key, sheet.getLastRow());
  }

  // 후처리(순위/정렬/색/정열)
  calculateRanking_(sheet);
  sortByRanking_(sheet);
  applyRankingColors_(sheet);
  applyAlignments_(sheet);

  return { inserted };
}

function num_(v) {
  const n = Number(v);
  return isNaN(n) ? 0 : n;
}

// 순위 계산(백카운트)
function calculateRanking_(sheet) {
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return;

  const players = [];
  for (let i = 1; i < data.length; i++) {
    players.push({
      row: i + 1,
      total: Number(data[i][6]) || 0, // G
      D: Number(data[i][10]) || 0,    // K
      C: Number(data[i][9]) || 0,     // J
      B: Number(data[i][8]) || 0,     // I
      A: Number(data[i][7]) || 0      // H
    });
  }

  players.sort((a, b) => {
    if (a.total !== b.total) return a.total - b.total;
    if (a.D !== b.D) return a.D - b.D;
    if (a.C !== b.C) return a.C - b.C;
    if (a.B !== b.B) return a.B - b.B;
    if (a.A !== b.A) return a.A - b.A;
    return 0;
  });

  let rank = 1;
  for (let i = 0; i < players.length; i++) {
    if (i > 0) {
      const prev = players[i - 1];
      const curr = players[i];
      if (
        prev.total !== curr.total ||
        prev.D !== curr.D ||
        prev.C !== curr.C ||
        prev.B !== curr.B ||
        prev.A !== curr.A
      ) rank = i + 1;
    }
    sheet.getRange(players[i].row, 1).setValue(rank);
  }
}

function sortByRanking_(sheet) {
  const lastRow = sheet.getLastRow();
  if (lastRow <= 1) return;

  const range = sheet.getRange(2, 1, lastRow - 1, 16);
  range.sort([
    { column: 1, ascending: true },  // 순위
    { column: 7, ascending: true },  // 합계
    { column: 11, ascending: true }  // D코스
  ]);
}

function applyRankingColors_(sheet) {
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return;

  const dataRange = sheet.getRange(2, 1, data.length - 1, 16);
  dataRange.setBackground('#ffffff')
    .setFontWeight('normal')
    .setFontSize(10)
    .setFontColor('#000000');

  for (let i = 1; i < data.length; i++) {
    const rank = data[i][0];
    if (rank === 1) {
      sheet.getRange(i + 1, 1, 1, 16).setBackground('#ffd700').setFontWeight('bold').setFontSize(11);
    } else if (rank === 2) {
      sheet.getRange(i + 1, 1, 1, 16).setBackground('#c0c0c0').setFontWeight('bold').setFontSize(11);
    } else if (rank === 3) {
      sheet.getRange(i + 1, 1, 1, 16).setBackground('#cd7f32').setFontWeight('bold').setFontSize(11);
    }
  }

  for (let r = 2; r <= data.length; r++) {
    sheet.getRange(r, 7).setBackground('#fff9c4').setFontWeight('bold');
  }

  for (let i = 1; i < data.length; i++) {
    const holeInOne = Number(data[i][11]) || 0;
    if (holeInOne > 0) {
      sheet.getRange(i + 1, 12).setBackground('#ffcccc').setFontWeight('bold').setFontColor('#cc0000');
    }
  }
}

// 정열: 숫자 우측 / 문자 중앙
function applyAlignments_(sheet) {
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return;

  const numCols = [1, 7, 8, 9, 10, 11, 12];
  const textCols = [2, 3, 4, 5, 6, 13, 14, 15, 16];

  numCols.forEach(c => sheet.getRange(2, c, lastRow - 1, 1).setHorizontalAlignment('right'));
  textCols.forEach(c => sheet.getRange(2, c, lastRow - 1, 1).setHorizontalAlignment('center'));
}

function copyMasterTemplate() {
    const urlInput = document.getElementById('masterTemplateUrl');
    const url = urlInput.value.trim();
    
    // URL 검증
    if (!url) {
        alert('⚠️ 마스터 템플릿 URL을 입력해주세요!');
        urlInput.focus();
        return;
    }
    
    // Google Sheets URL 검증
    if (!url.includes('docs.google.com/spreadsheets')) {
        alert('❌ 올바른 구글 시트 URL이 아닙니다!\n\n' +
              '예시: https://docs.google.com/spreadsheets/d/...');
        urlInput.focus();
        return;
    }
    
    // 시트 ID 추출
    let sheetId = '';
    
    // 패턴 1: /d/[ID]/edit
    const pattern1 = /\/d\/([a-zA-Z0-9-_]+)/;
    const match1 = url.match(pattern1);
    
    if (match1) {
        sheetId = match1[1];
    } else {
        alert('❌ URL에서 시트 ID를 찾을 수 없습니다.\n\n' +
              '올바른 URL 형식:\n' +
              'https://docs.google.com/spreadsheets/d/[시트ID]/edit');
        return;
    }
    
    // 복사 URL 생성
    const copyUrl = `https://docs.google.com/spreadsheets/d/${sheetId}/copy`;
    
    // 확인 메시지
    const confirmed = confirm(
        '📋 마스터 템플릿 복사\n\n' +
        '✅ 새 창에서 복사본이 생성됩니다.\n\n' +

    if (!confirmed) {
        return;

              '새 창에서 복사본이 열렸습니다.\n\n' +

/**
 * WebApp 엔드포인트
 */
function doPost(e) {
  try {
    const raw = (e && e.postData && e.postData.contents) ? e.postData.contents : '';
    const data = JSON.parse(raw || '{}');

    const incomingCode = (data.code || '').toString().trim().toUpperCase();
    const myCode = MY_CODE.toString().trim().toUpperCase();

    if (!incomingCode) throw new Error('대회 코드가 없습니다.');
    if (incomingCode !== myCode) {
      throw new Error('대회 코드가 일치하지 않습니다.\n\n입력: ' + incomingCode + '\n설정: ' + myCode);
    }

    const tournament = data.tournament || '대회명없음';
    const team = (data.team || data.teamName || data.jo || data.joName || '').toString().trim();
    const location = data.location || '장소없음';
    const timeStr = data.time || new Date().toLocaleString('ko-KR', { timeZone: 'Asia/Seoul' });

    const players = Array.isArray(data.players) ? data.players : [];
    if (!team) throw new Error('조명(팀명)이 없습니다.');
    if (players.length === 0) throw new Error('선수 데이터가 없습니다.');

    // ✅ 본선 여부 판단: data.stage / data.round / data.preliminaryDate 중 하나라도 "본선"이면 본선 처리
    const stageRaw = (data.stage || data.round || data.preliminaryDate || data.kind || '').toString().trim();
    const isFinal = stageRaw === '본선';

    const targetSheetName = isFinal ? SHEET_FINAL : SHEET_SCORE;
    const tag = isFinal ? '본선' : '점수기록';

    const result = insertOnlyToScoreSheet_(
      targetSheetName,
      incomingCode,
      tournament,
      team,
      location,
      tag,
      timeStr,
      players
    );

    // ✅ 데이터 들어올 때마다 4개 탭 자동맞춤
    autoFitCoreSheets_();

    return ContentService
      .createTextOutput(JSON.stringify({
        success: true,
        message: '전송 완료! (' + tag + ' 신규 ' + result.inserted + '건)\n※ 재전송(수정) 불가'
      }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({
        success: false,
        message: (err && err.message) ? err.message : String(err)
      }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function getSpreadsheet_() {
  if (SPREADSHEET_ID && SPREADSHEET_ID.toString().trim()) {
    return SpreadsheetApp.openById(SPREADSHEET_ID.toString().trim());
  }
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  if (!ss) {
    throw new Error(
      '스프레드시트를 찾을 수 없습니다.\n\n해결 방법:\n' +
      '1) 이 스크립트를 구글시트에 "바인드(연결)"해서 사용하거나\n' +
      '2) 코드 상단 SPREADSHEET_ID에 구글시트 ID를 입력하세요.'
    );
  }
  return ss;
}

/**
 * ✅ 점수기록/본선 시트 생성(없으면 자동 생성)
 */
function ensureScoreSheet_(ss, sheetName) {
  let sh = ss.getSheetByName(sheetName);
  if (sh) return sh;

  sh = ss.insertSheet(sheetName);
  sh.getRange(1, 1, 1, SCORE_HEADERS.length).setValues([SCORE_HEADERS]);

  const headerRange = sh.getRange(1, 1, 1, SCORE_HEADERS.length);
  headerRange
    .setFontWeight('bold')
    .setBackground('#4285f4')
    .setFontColor('#ffffff')
    .setHorizontalAlignment('center')
    .setVerticalAlignment('middle');

  sh.setFrozenRows(1);

  // 기본 폭(초기값)
  const widths = {
    1: 60,   2: 120,  3: 160,  4: 140,
    5: 70,   6: 120,  7: 70,   8: 70,
    9: 70,   10: 70,  11: 70,  12: 70,
    13: 180, 14: 120, 15: 120, 16: 90
  };
  Object.keys(widths).forEach(k => sh.setColumnWidth(Number(k), widths[k]));

  return sh;
}

/**
 * ✅ (정책) 재전송(수정) 절대 금지: append only
 * 중복키: 대회코드 + 조명 + 선수명 + 구분(점수기록/본선)
 */
function insertOnlyToScoreSheet_(sheetName, code, tournament, team, location, tag, timeStr, players) {
  const ss = getSpreadsheet_();
  const sh = ensureScoreSheet_(ss, sheetName);

  // 기존 데이터 인덱스 맵(중복 체크)
  const lastRow = sh.getLastRow();
  const existing = new Set();

  if (lastRow >= 2) {
    const values = sh.getRange(2, 1, lastRow - 1, 16).getValues();
    for (let i = 0; i < values.length; i++) {
      const row = values[i];
      const exCode = (row[1] || '').toString().trim();
      const exTeam = (row[2] || '').toString().trim();
      const exName = (row[3] || '').toString().trim();
      const exTag  = (row[15] || '').toString().trim();
      if (!exCode || !exName) continue;
      existing.add([exCode, exTeam, exName, exTag].join('|'));
    }
  }

  let inserted = 0;

  for (const p of players) {
    const name = (p && p.name ? p.name : '').toString().trim();
    if (!name) continue;

    const gender = (p.gender || p.sex || '').toString().trim();
    const affiliation = (p.affiliation || p.affil || p.org || '').toString().trim();

    if (!gender) throw new Error('성별이 비어 있습니다. 남/여 선택 후 다시 시도하세요.');

    const key = [code, team, name, tag].join('|');
    if (existing.has(key)) throw new Error('이미 전송된 기록입니다.\n수정이 필요하면 관리자에게 문의 하세요.');

    const rowValues = [
      '',                 // A 순위
      code,               // B 대회코드
      team,               // C 조명
      name,               // D 선수명
      gender,             // E 성별
      affiliation,        // F 소속
      num_(p.total),      // G 합계
      num_(p.A),          // H A코스
      num_(p.B),          // I B코스
      num_(p.C),          // J C코스
      num_(p.D),          // K D코스
      num_(p.holeInOne),  // L 홀인원
      timeStr,            // M 전송시간
      tournament,         // N 대회명
      location,           // O 장소
      tag                 // P 구분(점수기록/본선)
    ];

    sh.appendRow(rowValues);
    inserted++;
    existing.add(key);
  }

  // 후처리
  calculateRanking_(sh);
  sortByRanking_(sh);
  applyRankingColors_(sh);
  applyAlignments_(sh);
  applyColumnSizingAndFonts_(sh);

  return { inserted };
}

function num_(v) {
  const n = Number(v);
  return isNaN(n) ? 0 : n;
}

// 순위 계산(백카운트: 합계 → D → C → B → A)
function calculateRanking_(sheet) {
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return;

  const arr = [];
  for (let i = 1; i < data.length; i++) {
    arr.push({
      row: i + 1,
      total: Number(data[i][6]) || 0, // G
      D: Number(data[i][10]) || 0,    // K
      C: Number(data[i][9]) || 0,     // J
      B: Number(data[i][8]) || 0,     // I
      A: Number(data[i][7]) || 0      // H
    });
  }

  arr.sort((a, b) => {
    if (a.total !== b.total) return a.total - b.total;
    if (a.D !== b.D) return a.D - b.D;
    if (a.C !== b.C) return a.C - b.C;
    if (a.B !== b.B) return a.B - b.B;
    if (a.A !== b.A) return a.A - b.A;
    return 0;
  });

  let rank = 1;
  for (let i = 0; i < arr.length; i++) {
    if (i > 0) {
      const p = arr[i - 1];
      const c = arr[i];
      if (p.total !== c.total || p.D !== c.D || p.C !== c.C || p.B !== c.B || p.A !== c.A) {
        rank = i + 1;
      }
    }
    sheet.getRange(arr[i].row, 1).setValue(rank);
  }
}

function sortByRanking_(sheet) {
  const lastRow = sheet.getLastRow();
  if (lastRow <= 1) return;

  sheet.getRange(2, 1, lastRow - 1, 16).sort([
    { column: 1, ascending: true },
    { column: 7, ascending: true },
    { column: 11, ascending: true }
  ]);
}

function applyRankingColors_(sheet) {
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return;

  const range = sheet.getRange(2, 1, data.length - 1, 16);
  range.setBackground('#ffffff').setFontWeight('normal').setFontSize(10).setFontColor('#000000');

  for (let i = 1; i < data.length; i++) {
    const rank = data[i][0];
    if (rank === 1) sheet.getRange(i + 1, 1, 1, 16).setBackground('#ffd700').setFontWeight('bold').setFontSize(11);
    else if (rank === 2) sheet.getRange(i + 1, 1, 1, 16).setBackground('#c0c0c0').setFontWeight('bold').setFontSize(11);
    else if (rank === 3) sheet.getRange(i + 1, 1, 1, 16).setBackground('#cd7f32').setFontWeight('bold').setFontSize(11);
  }

  // 합계 강조
  for (let r = 2; r <= data.length; r++) {
    sheet.getRange(r, 7).setBackground('#fff9c4').setFontWeight('bold');
  }

  // 홀인원 강조
  for (let i = 1; i < data.length; i++) {
    const hole = Number(data[i][11]) || 0;
    if (hole > 0) {
      sheet.getRange(i + 1, 12).setBackground('#ffcccc').setFontWeight('bold').setFontColor('#cc0000');
    }
  }
}

// 정열: 숫자 우측 / 문자 중앙
function applyAlignments_(sheet) {
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return;

  const numCols = [1, 7, 8, 9, 10, 11, 12];
  const textCols = [2, 3, 4, 5, 6, 13, 14, 15, 16];

  numCols.forEach(c => sheet.getRange(2, c, lastRow - 1, 1).setHorizontalAlignment('right'));
  textCols.forEach(c => sheet.getRange(2, c, lastRow - 1, 1).setHorizontalAlignment('center'));
}

function applyColumnSizingAndFonts_(sheet) {
  const lastRow = sheet.getLastRow();
  const lastCol = sheet.getLastColumn();
  if (lastCol < 1) return;

  // 헤더/데이터 폰트
  if (lastRow >= 1) sheet.getRange(1, 1, 1, lastCol).setFontWeight('bold');
  if (lastRow >= 2) sheet.getRange(2, 1, lastRow - 1, lastCol).setFontWeight('normal');

  // ✅ 1) 우선 전체 자동맞춤(한번만)
  sheet.autoResizeColumns(1, lastCol);

  // ✅ 2) 짧은 열은 "고정폭"으로 강제 (헤더가 넓어지는 문제 해결)
  // A~O(15열) 기준 (현재 시트 구조와 동일)
  const fixed = {
    1: 45,   // 순위
    2: 95,   // 대회코드
    3: 85,   // 조명
    4: 85,   // 선수명
    5: 45,   // 성별
    6: 70,   // 소속
    7: 55,   // 합계
    8: 55,   // A코스
    9: 55,   // B코스
    10: 55,  // C코스
    11: 55,  // D코스
    12: 55,  // 홀인원
    13: 160, // 전송시간 (너무 길면 140~180 사이 조절)
    14: 90,  // 대회명
    15: 80   // 장소
    // 16열(구분)이 있으면 16:70 추가
  };

  Object.keys(fixed).forEach(k => {
    const col = Number(k);
    if (col <= lastCol) sheet.setColumnWidth(col, fixed[col]);
  });

  // ✅ 3) 너무 좁아지는 경우만 최소폭 보정(옵션)
  const minWidth = { 2: 85, 3: 75, 4: 75, 13: 140 };
  Object.keys(minWidth).forEach(k => {
    const col = Number(k);
    if (col <= lastCol && sheet.getColumnWidth(col) < minWidth[col]) {
      sheet.setColumnWidth(col, minWidth[col]);
    }
  });
}


/* -----------------------------
   Auto-fit core sheets (always)
------------------------------ */

function autoFitCoreSheets_() {
  const ss = getSpreadsheet_();
  const core = [SHEET_REG, SHEET_GROUP, SHEET_SCORE, SHEET_FINAL];

  core.forEach(name => {
    const sh = ss.getSheetByName(name);
    if (!sh) return;

    try {
      const lastCol = sh.getLastColumn();
      if (lastCol < 1) return;

      sh.autoResizeColumns(1, lastCol);

      // 점수 탭은 최소폭 보정까지 적용
      if (name === SHEET_SCORE || name === SHEET_FINAL) {
        applyColumnSizingAndFonts_(sh);
      }
    } catch (e) {
      // 보호/권한 문제 시 해당 탭만 스킵
    }
  });
}

// 수동 실행용(지금 당장 4개 탭 자동맞춤)
function RUN_AUTO_FIT_NOW() {
  autoFitCoreSheets_();
}
